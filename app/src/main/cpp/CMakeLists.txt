cmake_minimum_required(VERSION 3.22)
project(cinator_quickjs LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# app/src/main/cpp -> repository root is exactly four parents.
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(QUICKJS_DIR "${REPOSITORY_ROOT}/vendor/quickjs")
set(QUICKJS_COMMIT "04be246001599f5995fa2f2d8c91a0f198d3f34c")
if(NOT EXISTS "${QUICKJS_DIR}/PROVENANCE.json" OR NOT EXISTS "${QUICKJS_DIR}/VERSION")
    message(FATAL_ERROR "Pinned vendor/quickjs source is missing")
endif()
file(READ "${QUICKJS_DIR}/PROVENANCE.json" PROVENANCE)
string(JSON VENDOR_COMMIT GET "${PROVENANCE}" commit)
file(STRINGS "${QUICKJS_DIR}/VERSION" VENDOR_VERSION LIMIT_COUNT 1)
if(NOT VENDOR_COMMIT STREQUAL QUICKJS_COMMIT OR NOT VENDOR_VERSION STREQUAL "2026-06-04")
    message(FATAL_ERROR "QuickJS source pin mismatch")
endif()
string(JSON FILE_COUNT LENGTH "${PROVENANCE}" files)
math(EXPR LAST_FILE "${FILE_COUNT} - 1")
foreach(INDEX RANGE 0 ${LAST_FILE})
    string(JSON NAME GET "${PROVENANCE}" files ${INDEX} file)
    string(JSON EXPECTED_HASH GET "${PROVENANCE}" files ${INDEX} sha256)
    if(NOT EXISTS "${QUICKJS_DIR}/${NAME}")
        message(FATAL_ERROR "Pinned QuickJS file missing: ${NAME}")
    endif()
    file(SHA256 "${QUICKJS_DIR}/${NAME}" ACTUAL_HASH)
    if(NOT ACTUAL_HASH STREQUAL EXPECTED_HASH)
        message(FATAL_ERROR "Pinned QuickJS file mismatch: ${NAME}")
    endif()
endforeach()

add_library(quickjs_core STATIC
    "${QUICKJS_DIR}/quickjs.c" "${QUICKJS_DIR}/dtoa.c"
    "${QUICKJS_DIR}/libregexp.c" "${QUICKJS_DIR}/libunicode.c" "${QUICKJS_DIR}/cutils.c")
# Upstream uses GNU asm in its Atomics implementation; keep our JNI bridge strict C11.
set_target_properties(quickjs_core PROPERTIES POSITION_INDEPENDENT_CODE ON C_EXTENSIONS ON)
target_include_directories(quickjs_core SYSTEM PUBLIC "${QUICKJS_DIR}")
target_compile_definitions(quickjs_core PUBLIC _GNU_SOURCE CONFIG_VERSION="2026-06-04")
add_library(cinator_quickjs SHARED cinator_quickjs.c)
target_link_libraries(cinator_quickjs PRIVATE quickjs_core m)
target_compile_options(cinator_quickjs PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
    target_link_options(cinator_quickjs PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
else()
    find_package(JNI REQUIRED)
    find_package(Threads REQUIRED)
    target_include_directories(cinator_quickjs PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(cinator_quickjs PRIVATE Threads::Threads)
    include(CTest)
    if(BUILD_TESTING)
        add_subdirectory("${REPOSITORY_ROOT}/app/src/test/native" "native-tests")
    endif()
endif()
